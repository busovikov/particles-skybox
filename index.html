your-world.js
import { Timer } from 'three/webgpu';
import { ParticleSkybox } from 'particles-skybox';

const sky = await ParticleSkybox.create({
  renderer,  // your WebGPURenderer
  scene,     // your existing 3D world
  camera,
  warmup: 12 // simulation seconds
});

const timer = new Timer();
renderer.setAnimationLoop(timestamp => {
  timer.update(timestamp);
  sky.update(timer.getDelta(), camera);
  renderer.render(scene, camera);
});

Skip the cold start.

Warm up before the first frame, with progress and cancellation. No intermediate cube captures.
